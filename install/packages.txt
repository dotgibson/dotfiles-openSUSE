# dotfiles-openSUSE/install/packages.txt
# zypper package names for the modern terminal stack. One per line; # comments ok.
# Installed by bootstrap.sh instead of listed here — keep them OUT: starship, atuin,
# yazi, viddy (upstream installer / cargo, each presence-guarded, so a packaged binary
# just short-circuits the install). Tumbleweed's repo-oss now ships those four
# first-class; they stay out to keep the upstream-latest build, not for lack of a
# package. tree-sitter-cli is the one that really isn't packaged — cargo only.

# ── shell + multiplexer ───────────────────────────────────────
zsh
tmux

# ── modern CLI core ───────────────────────────────────────────
eza            # ls
bat            # cat
fd             # find  (openSUSE packages it as `fd`, binary is `fd` — not fdfind)
ripgrep        # grep
zoxide         # cd
fzf            # fuzzy finder
git-delta      # git pager / diffs (binary: delta)
btop           # top/htop
tealdeer       # tldr (binary: tldr). Tumbleweed + Leap 15.6 OSS; older Leap: cargo install tealdeer

# ── modern CLI extras (core-doctor set) ───────────────────────
# All eight verified against the openSUSE OSS index (2026-08-08, #83) — each resolves
# under the name written, so the old "verify on-box" hedge here is discharged. The
# resilient zypper_install in bootstrap.sh still falls back to per-package installs, so
# a name that's absent on an older Leap is skipped rather than aborting the run.
dust           # du (disk usage)
duf            # df (disk free)
procs          # ps
sd             # sed (find/replace)
gron           # greppable JSON
glow           # markdown renderer
gum            # shell-script UI toolkit
xh             # HTTPie-like (HTTP client)
# Installed by bootstrap.sh, not packaged here:
#   - go-installed: yq (mikefarah), doggo (DNS), carapace (completions), sesh (tmux)
#   - vendor repo:  op (1Password CLI, from 1Password's signed rpm repo)
# yq is the mikefarah Go v4 build: stock `zypper in yq` does NOT reliably provide it —
# the main OSS `yq` is kislyuk's SEPARATE `python-yq` (a different tool), and mikefarah's
# Go build only ships from a personal OBS repo — so it's go-installed like doggo/carapace.

# ── editor + git workflow ─────────────────────────────────────
neovim
git              # `git subtree` ships WITH git on openSUSE (no separate package, unlike Fedora)
openssh-clients  # ssh / ssh-keygen / ssh-agent. Older Leap may call this just `openssh`
lazygit          # in openSUSE repos (unlike Fedora, which needs a COPR)

# ── clipboard (Wayland-first; inert under WSL where clip.exe is used) ─────────
wl-clipboard   # wl-copy / wl-paste
xclip          # X11 fallback (sessions / SSH X-forward)

# ── build + language toolchain (needed for cargo-installed tools) ─
gcc
gcc-c++        # some tree-sitter parsers ship C++ scanners
make
cargo          # rust toolchain -> builds yazi + tree-sitter-cli

# ── Neovim provider deps (fix :checkhealth) ───────────────────
python3-neovim # pynvim — the Python 3 provider module
python3-pip

curl
wget

htop
